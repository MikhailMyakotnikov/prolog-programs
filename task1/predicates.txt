size(x,y) - файл x имеет размер y
delete(x) - файл x удаляется сразу
move(x,y) - файл x перемещается в папку y
store(x,y) - файл x хранится в папке y

move(File, kor),size(File,large) -> delete(File)
move(File, Dir),Dir \= kor -> store(File, Dir)
move(File, kor)size(File,small) -> store(File, kor)

size(script, small)
size(video, large)
size(text, small)
move(script, kor)
move(video, kor)
move(text, docs)

store(X, kor).
delete(X).
store(X, docs).
